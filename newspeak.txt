function speakSashaReply(text){
  const clean = String(text||'').replace(/\s+/g,' ').trim();
  if(!clean) return;
  if(window.sashaAudio){ try{ window.sashaAudio.pause(); }catch(e){} }
  if(window.speechSynthesis) window.speechSynthesis.cancel();
  fetch('https://fincop-backend.onrender.com/sasha-speak', {
    method:'POST', headers:{'Content-Type':'application/json'},
    body: JSON.stringify({ text: clean })
  }).then(function(r){
    if(!r.ok) throw new Error('tts ' + r.status);
    return r.blob();
  }).then(function(b){
    window.sashaAudio = new Audio(URL.createObjectURL(b));
    window.sashaAudio.play();
  }).catch(function(e){
    console.error('[sasha voice]', e);
    if(window.speechSynthesis) window.speechSynthesis.speak(new SpeechSynthesisUtterance(clean));
  });
}
